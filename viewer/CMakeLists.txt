function(pvd_viewer_copy_runtime target)
    if(BUILD_SHARED_LIBS)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:pvd_core> $<TARGET_FILE:pvd_render> $<TARGET_FILE_DIR:${target}>)
        if(PVD_BUILD_TRANSPORT)
            add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_FILE:pvd_transport> $<TARGET_FILE_DIR:${target}>)
        endif()
        if(PVD_BUILD_ANALYSIS)
            add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_FILE:pvd_analysis> $<TARGET_FILE_DIR:${target}>)
        endif()
        if(PVD_BUILD_REPX)
            add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                $<TARGET_FILE:pvd_repx> $<TARGET_FILE_DIR:${target}>)
        endif()
    endif()
endfunction()
