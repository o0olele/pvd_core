cmake_minimum_required(VERSION 3.20)
project(pvd_external_consumer LANGUAGES CXX)
option(PVD_CONSUMER_TRANSPORT "Also exercise the installed TCP module" OFF)
option(PVD_CONSUMER_RENDER "Also exercise the installed rendering extraction module" OFF)
find_package(pvd_core 0.1 CONFIG REQUIRED)
add_executable(pvd_external_consumer main.cpp)
target_link_libraries(pvd_external_consumer PRIVATE pvd::core)
if(PVD_CONSUMER_TRANSPORT)
    target_link_libraries(pvd_external_consumer PRIVATE pvd::transport)
    target_compile_definitions(pvd_external_consumer PRIVATE PVD_CONSUMER_TRANSPORT)
endif()
if(PVD_CONSUMER_RENDER)
    target_link_libraries(pvd_external_consumer PRIVATE pvd::render)
    target_compile_definitions(pvd_external_consumer PRIVATE PVD_CONSUMER_RENDER)
endif()
